create or replace function public.spend_credits(p_user_id uuid, p_amount integer)
returns table (success boolean, credits_after integer)
language plpgsql security definer set search_path = public as $$
declare bal integer;
begin
  if p_amount is null or p_amount <= 0 then
    return query select true, p.credits from public.profiles p where p.id = p_user_id; return;
  end if;
  update public.profiles set credits = credits - p_amount
    where id = p_user_id and credits >= p_amount returning credits into bal;
  if bal is null then
    return query select false, coalesce((select credits from public.profiles where id = p_user_id), 0);
  else return query select true, bal; end if;
end; $$;

create or replace function public.add_credits(p_user_id uuid, p_amount integer)
returns table (success boolean, credits_after integer)
language plpgsql security definer set search_path = public as $$
declare bal integer;
begin
  if p_amount is null or p_amount <= 0 then
    return query select true, p.credits from public.profiles p where id = p_user_id; return;
  end if;
  update public.profiles set credits = credits + p_amount where id = p_user_id returning credits into bal;
  return query select (bal is not null), coalesce(bal, 0);
end; $$;

revoke execute on function public.spend_credits(uuid, integer) from public, anon, authenticated;
revoke execute on function public.add_credits(uuid, integer) from public, anon, authenticated;
grant execute on function public.spend_credits(uuid, integer) to service_role;
grant execute on function public.add_credits(uuid, integer) to service_role;

create table if not exists public.processed_stripe_events (
  event_id text primary key, event_type text not null,
  processed_at timestamptz not null default now());
alter table public.processed_stripe_events enable row level security;

C2. THE "~21 FACTORS" THAT DRIVE THE LOCAL "near me" PACK -------------------
Google ranks the local 3-pack / map results mainly on RELEVANCE, DISTANCE,
PROMINENCE. You cannot *force* a ranking; you can only optimize the controllable
signals below and measure them. These are the checklist our scorer will grade:

  PROFILE COMPLETENESS
   1. Verified listing + exact, consistent NAP (Name/Address/Phone)
   2. Correct PRIMARY category (single biggest ranking lever)
   3. Relevant SECONDARY categories
   4. Clean business name (NO keyword stuffing -> can get suspended)
   5. Full description with natural keywords
   6. Services AND/OR products added (menu/services)
   7. Attributes ticked (accessibility, payments, "online appointments", etc.)
   8. Accurate standard hours + special (holiday) hours
   9. Correct map pin / service areas
  CONTENT & ENGAGEMENT
  10. Photo count + quality (exterior, interior, team, product)
  11. Photo RECENCY (fresh uploads help)
  12. Videos present
  13. Google Posts cadence (regular updates/offers/events)
  14. Website link (+ booking/appointment URL with UTM tracking)
  15. Phone + chat/messaging enabled
  16. Q&A section populated with owner answers
  REVIEW SIGNALS
  17. Number of reviews (volume)
  18. Average star rating
  19. Review RECENCY / velocity
  20. Owner reply RATE (every review answered?)
  21. Owner reply TIME (responded quickly?) + keyword-rich, varied replies
  OFF-PROFILE (measured, not directly editable via API)
  +  Citations/NAP consistency across directories
  +  On-page + technical SEO of the linked WEBSITE (see C3)
  +  Backlinks / prominence, proximity to searcher

  For "hotel near me" / "optical near me": the map 3-pack comes first (factors
  1-21 above), then 7 organic web results ranked by normal website SEO. So a
  complete GBP + an SEO-healthy website together win the page.

C3. HOW WE WOULD COMPUTE THE "OPTIMIZATION %" (SEO SCORE) -------------------
No single Google API returns a "score". We build our own heuristic scorer that:
  a. Pulls the profile via the GBP APIs -> checks each of the 21 factors
  b. Audits the linked WEBSITE (server fetch + parse) for:
       - <title> length (50-60), meta description (120-160), single H1
       - image alt text, canonical tag, mobile viewport, HTTPS
       - sitemap.xml + robots.txt present
       - LocalBusiness / Hotel / Optical schema.org structured data (JSON-LD)
       - Local landing page (NAP shown, embedded map, area-service copy)
  c. Optionally calls Google PageSpeed Insights / Lighthouse API for the
     Performance + SEO sub-scores (0-100 each).
  Weight the factors -> output a 0-100% "Profile Optimization" score + a ranked
  to-do list ("you have 3 photos, need 10+", "reply to 12 unanswered reviews",
  "missing LocalBusiness schema", etc.).
